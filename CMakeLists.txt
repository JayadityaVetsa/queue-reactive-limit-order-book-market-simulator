cmake_minimum_required(VERSION 3.16)
project(lob_execution LANGUAGES CXX)
add_library(execution SHARED native/execution.cpp)
target_compile_features(execution PRIVATE cxx_std_17)
set_target_properties(execution PROPERTIES PREFIX "" OUTPUT_NAME "execution")
if(MSVC)
  target_compile_options(execution PRIVATE /W4 /WX)
else()
  target_compile_options(execution PRIVATE -Wall -Wextra -Werror)
endif()
